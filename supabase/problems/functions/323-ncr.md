In how many ways can you choose `r` items from `n`? Write `unsigned long long ncr(int n, int r)` and answer `t` queries.

Computing factorials overflows quickly, so use the multiplicative formula: multiply by (n − r + i) and divide by i for i = 1 to r. Each step stays a whole number.

**Input:** `t`, then `t` lines each with `n r`.

**Output:** nCr for each query.

**Constraints:** 1 ≤ t ≤ 1000, 0 ≤ r ≤ n ≤ 62

## Hints

1. Use symmetry: nCr = nC(n − r), so set `r = min(r, n − r)` to keep the loop short.
2. In the loop, `result = result * (n - r + i) / i`. Multiply before dividing; the division is always exact.

```c starter
#include <stdio.h>

unsigned long long ncr(int n, int r) {
    return 1;
}

int main(void) {
    int t;
    scanf("%d", &t);
    while (t--) {
        int n, r;
        scanf("%d %d", &n, &r);
        printf("%llu\n", ncr(n, r));
    }
    return 0;
}
```

```c solution
#include <stdio.h>

unsigned long long ncr(int n, int r) {
    if (r > n - r) r = n - r;
    unsigned long long result = 1;
    for (int i = 1; i <= r; i++) result = result * (unsigned long long)(n - r + i) / (unsigned long long)i;
    return result;
}

int main(void) {
    int t;
    if (scanf("%d", &t) != 1) return 1;
    while (t--) {
        int n, r;
        if (scanf("%d %d", &n, &r) != 2) return 1;
        printf("%llu\n", ncr(n, r));
    }
    return 0;
}
```
